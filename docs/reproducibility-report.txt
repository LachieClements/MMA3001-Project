    note("Only temporary repositories receive fixture commits/configuration. Project index and HEAD remain untouched.")
    note("All child processes remove PYTHONPATH/PYTHONHOME. No sys.path injection is used.")
    tracked = subprocess.check_output(
        ["git", "ls-files", "-z"], cwd=PROJECT, env=clean_environment(),
    ).split(b"\0")
    names = sorted(set([item.decode("utf-8") for item in tracked if item] + [".gitattributes"]))
    FIXTURE.mkdir()
    file_hashes = {}
    for name in names:
        source = PROJECT / name
        assert source.is_file() and source.resolve().is_relative_to(PROJECT.resolve())
        assert not ({".git", ".venv", ".aws", ".codex", ".agents"} & set(Path(name).parts))
        target = FIXTURE / name
        target.parent.mkdir(parents=True, exist_ok=True)
        content = source.read_bytes()
        target.write_bytes(content)
        file_hashes[name] = sha256(content).hexdigest()
    (BASE / "fixture-files.json").write_text(json.dumps(file_hashes, indent=2), encoding="utf-8")
    run(["git", "--version"], BASE, "git-version")
    run(["git", "init", "--initial-branch=main", FIXTURE], BASE, "fixture-init")
    run(["git", "-C", FIXTURE, "config", "core.autocrlf", "true"], BASE, "fixture-autocrlf")
    run(["git", "-C", FIXTURE, "add", "--all"], BASE, "fixture-add")
    run([
        "git", "-C", FIXTURE, "-c", "user.name=MMA3001 reproduction fixture",
        "-c", "user.email=fixture@example.invalid", "-c", "commit.gpgsign=false",
        "-c", "core.hooksPath=" + str(BASE / "no-hooks"),
        "commit", "-m", "Temporary Milestone 1 reproducibility fixture",
    ], BASE, "temporary-fixture-commit")
    for name in (
        "data/raw/airfoil_self_noise.dat",
        "data/splits/conditions.csv",
        "data/splits/development_folds.csv",
    ):
        blob = run(["git", "-C", FIXTURE, "show", "HEAD:" + name], BASE, "committed-bytes-" + name, binary=True)
        assert sha256(blob).hexdigest() == BEFORE["protected_files"][name]
    run(["git", "clone", "--no-checkout", "--no-local", FIXTURE, CHECKOUT], BASE, "isolated-clone")
    run(["git", "-C", CHECKOUT, "config", "core.autocrlf", "true"], BASE, "clone-autocrlf-before-checkout")
    run(["git", "-C", CHECKOUT, "checkout", "main"], BASE, "fresh-checkout")
    run(["git", "-C", CHECKOUT, "config", "--show-origin", "--get", "core.autocrlf"], BASE, "clone-autocrlf-proof")
    run([
        "git", "-C", CHECKOUT, "check-attr", "text", "eol", "--",
        "data/raw/airfoil_self_noise.dat", "data/splits/conditions.csv", "data/splits/development_folds.csv",
    ], BASE, "checkout-attributes")
    run([
        "git", "-C", CHECKOUT, "ls-files", "--eol", "--",
        "data/raw/airfoil_self_noise.dat", "data/splits/conditions.csv", "data/splits/development_folds.csv",
    ], BASE, "checkout-line-endings")
    assert not run(["git", "-C", CHECKOUT, "status", "--porcelain"], BASE, "clean-checkout").strip()
    check_checkout("Before installation")
    OUTSIDE.mkdir()
    run([sys.executable, "-m", "venv", ".venv"], CHECKOUT, "fresh-venv")
    note("Fresh pyvenv.cfg:\n" + (VENV / "pyvenv.cfg").read_text())
    run([PYTHON, "-m", "pip", "list", "--format=json"], CHECKOUT, "bootstrap-packages")
    assert_original_untouched()
